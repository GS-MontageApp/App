window.submitPinLogin = function() { const pinInput = document.getElementById('pin_input'); const pin = pinInput ? pinInput.value.trim() : ''; const headerEl = document.getElementById('main_header'); const badgeEl = document.getElementById('role_status_badge'); const logoutItem = document.getElementById('logout_menu_item'); if (headerEl) { headerEl.classList.remove('role-header-root', 'role-header-user'); } if (pin === '0633') { if (headerEl) headerEl.classList.add('role-header-root'); if (badgeEl) { badgeEl.textContent = '~/root'; badgeEl.className = 'badge-root'; badgeEl.classList.remove('hidden'); } if (logoutItem) logoutItem.classList.remove('hidden'); // Logout-Button einblenden localStorage.setItem('zangenschlosser_role', 'root'); window.closePinModal(); } else if (pin === '0449') { if (headerEl) headerEl.classList.add('role-header-user'); if (badgeEl) { badgeEl.textContent = '~/benutzer'; badgeEl.className = 'badge-user'; badgeEl.classList.remove('hidden'); } if (logoutItem) logoutItem.classList.remove('hidden'); // Logout-Button einblenden localStorage.setItem('zangenschlosser_role', 'user'); window.closePinModal(); } else { alert('Ungültige PIN!'); if (pinInput) { pinInput.value = ''; pinInput.focus(); } } }; window.logoutRole = function() { const headerEl = document.getElementById('main_header'); const badgeEl = document.getElementById('role_status_badge'); const logoutItem = document.getElementById('logout_menu_item'); if (headerEl) headerEl.classList.remove('role-header-root', 'role-header-user'); if (badgeEl) { badgeEl.textContent = ''; badgeEl.className = 'hidden'; } if (logoutItem) logoutItem.classList.add('hidden'); // Logout-Button ausblenden localStorage.removeItem('zangenschlosser_role'); };
